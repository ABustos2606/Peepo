cmake_minimum_required(VERSION 3.25)
project(Peepo VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

add_library(peepo_core src/core/model.cpp src/core/reference_catalogue.cpp src/engine/simulated_engine.cpp
  src/engine/ingen/turtle.cpp src/engine/ingen/protocol.cpp src/engine/ingen/session.cpp
  src/engine/ingen/socket_transport.cpp src/engine/ingen/ingen_engine.cpp
  src/catalogue/json.cpp src/catalogue/catalogue.cpp
  src/storage/bundle.cpp src/storage/patch_store.cpp src/storage/file_store.cpp src/controls/controls.cpp)
target_include_directories(peepo_core PUBLIC src)
find_package(Threads REQUIRED)
target_link_libraries(peepo_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(peepo_core PUBLIC ws2_32)
else()
  target_compile_options(peepo_core PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
endif()
if(MSVC)
  target_compile_options(peepo_core PRIVATE /W4 /permissive- /utf-8)
endif()

# Physical controls on Linux: the kernel's input devices (rotary encoders and
# GPIO keys on the Pi). No UI dependency, so it is tested without a display.
if(UNIX AND NOT APPLE)
  add_library(peepo_linux_input src/platform/linux/evdev_controls.cpp)
  target_link_libraries(peepo_linux_input PUBLIC peepo_core)
  target_compile_options(peepo_linux_input PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
  # The network links shown in Settings, so the instrument can be found, and
  # the machine's readings on the Diagnostics page.
  add_library(peepo_linux_network src/platform/linux/network_status.cpp src/platform/linux/system_status.cpp)
  target_include_directories(peepo_linux_network PUBLIC src)
  target_compile_options(peepo_linux_network PRIVATE -Wall -Wextra)
  # The instrument's own MIDI (clock out), a JACK client of its own.
  find_package(PkgConfig)
  if(PkgConfig_FOUND)
    pkg_check_modules(PEEPO_JACK IMPORTED_TARGET jack)
  endif()
  if(PEEPO_JACK_FOUND)
    add_executable(peepo-midi src/platform/linux/midi_main.cpp)
    target_link_libraries(peepo-midi PRIVATE peepo_core PkgConfig::PEEPO_JACK)
    target_compile_options(peepo-midi PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
    install(TARGETS peepo-midi RUNTIME DESTINATION bin)
  endif()
endif()

# The exporter reads installed plugins through Lilv, as the engine does.
# It is a Linux tool; its output is what other platforms use as a fixture.
if(NOT WIN32)
  find_package(PkgConfig)
  if(PkgConfig_FOUND)
    pkg_check_modules(LILV IMPORTED_TARGET lilv-0)
  endif()
  if(LILV_FOUND)
    add_executable(peepo_catalogue_export tools/catalogue-export/main.cpp)
    target_link_libraries(peepo_catalogue_export PRIVATE peepo_core PkgConfig::LILV)
    target_compile_definitions(peepo_catalogue_export PRIVATE PEEPO_LILV_VERSION="${LILV_VERSION}")
    target_compile_options(peepo_catalogue_export PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
    install(TARGETS peepo_catalogue_export RUNTIME DESTINATION bin)
  endif()
endif()

option(PEEPO_BUILD_UI "Build LVGL simulator and visual checks" ON)
if(PEEPO_BUILD_UI)
  include(FetchContent)
  set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/config/lv_conf.h" CACHE PATH "" FORCE)
  set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(lvgl
    GIT_REPOSITORY https://github.com/lvgl/lvgl.git
    GIT_TAG 85aa60d18b3d5e5588d7b247abf90198f07c8a63)
  FetchContent_MakeAvailable(lvgl)
  if(WIN32)
    include(cmake/lvgl_windows_fix.cmake)
    target_link_libraries(lvgl PUBLIC user32 gdi32 shell32)
    target_compile_definitions(lvgl PUBLIC NOMINMAX)
  endif()
  if(UNIX AND NOT APPLE)
    # The Pi draws through DRM/KMS (or the framebuffer) with evdev touch; a
    # desktop Linux, or WSL, can open an SDL window instead. Each is built when
    # its library is present, and chosen at run time with --display.
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PEEPO_DRM IMPORTED_TARGET libdrm)
    option(PEEPO_WITH_SDL "Build the SDL window for Linux desktops and WSL" ON)
    if(PEEPO_WITH_SDL)
      pkg_check_modules(PEEPO_SDL IMPORTED_TARGET sdl2)
    endif()
    if(PEEPO_DRM_FOUND)
      target_compile_definitions(lvgl PUBLIC PEEPO_LVGL_DRM)
      target_link_libraries(lvgl PUBLIC PkgConfig::PEEPO_DRM)
    endif()
    if(PEEPO_SDL_FOUND)
      target_compile_definitions(lvgl PUBLIC PEEPO_LVGL_SDL)
      target_link_libraries(lvgl PUBLIC PkgConfig::PEEPO_SDL)
    endif()
    message(STATUS "Linux displays: framebuffer, DRM=${PEEPO_DRM_FOUND}, SDL=${PEEPO_SDL_FOUND}")
  endif()
  add_library(peepo_ui src/ui/application.cpp
    src/ui/fonts/peepo_bold_14.c src/ui/fonts/peepo_bold_18.c src/ui/fonts/peepo_bold_20.c
    src/ui/fonts/peepo_bold_24.c src/ui/fonts/peepo_bold_48.c src/ui/fonts/peepo_mono_14.c)
  target_link_libraries(peepo_ui PUBLIC peepo_core lvgl)
  target_include_directories(peepo_ui PUBLIC src)
  if(MSVC)
    target_compile_options(peepo_ui PRIVATE /W4 /permissive- /utf-8)
  endif()
  if(WIN32)
    add_executable(peepo WIN32 src/platform/windows/main.cpp)
    target_link_libraries(peepo PRIVATE peepo_ui)
    target_compile_definitions(peepo PRIVATE UNICODE _UNICODE NOMINMAX)
    if(MSVC)
      target_compile_options(peepo PRIVATE /utf-8)
    endif()
    add_custom_command(TARGET peepo POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:peepo>/licenses"
      # What --engine offers by default: the exported fixture and the project's overlay.
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/config/catalogue/wsl-debian.json" "$<TARGET_FILE_DIR:peepo>/catalogue.json"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/config/catalogue/overlay.json" "$<TARGET_FILE_DIR:peepo>/catalogue-overlay.json")
    if(BUILD_TESTING)
      add_test(NAME windows_driver_smoke COMMAND peepo --smoke)
      set_tests_properties(windows_driver_smoke PROPERTIES TIMEOUT 15)
    endif()
  endif()
  if(UNIX AND NOT APPLE)
    add_executable(peepo src/platform/linux/main.cpp)
    target_link_libraries(peepo PRIVATE peepo_ui peepo_linux_input peepo_linux_network)
    target_compile_options(peepo PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
    add_custom_command(TARGET peepo POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/config/catalogue/wsl-debian.json" "$<TARGET_FILE_DIR:peepo>/catalogue.json"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/config/catalogue/overlay.json" "$<TARGET_FILE_DIR:peepo>/catalogue-overlay.json")
    install(TARGETS peepo RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
      add_test(NAME linux_display_smoke COMMAND peepo --display offscreen --smoke)
      set_tests_properties(linux_display_smoke PROPERTIES TIMEOUT 15)
    endif()
  endif()
  add_executable(peepo_ui_check tests/ui_check.cpp)
  target_link_libraries(peepo_ui_check PRIVATE peepo_ui)
  if(BUILD_TESTING)
    add_test(NAME ui_workflows COMMAND peepo_ui_check "${CMAKE_BINARY_DIR}/screenshots")
    # About 7 s natively; the arm64 build runs it under emulation, several times slower.
    set_tests_properties(ui_workflows PROPERTIES TIMEOUT 120)
  endif()
endif()

  target_link_libraries(peepo_output_level_live PRIVATE peepo_core)
  install(TARGETS peepo_output_level_live RUNTIME DESTINATION libexec/peepo)
  # A module's file (model, impulse response) through the adapter: see tools/pi/check-engine.sh.
  add_executable(peepo_file_live tests/file_live.cpp)
  target_link_libraries(peepo_file_live PRIVATE peepo_core)
  install(TARGETS peepo_file_live RUNTIME DESTINATION libexec/peepo)
  add_executable(peepo_catalogue_live tests/catalogue_live.cpp)
  target_link_libraries(peepo_catalogue_live PRIVATE peepo_core)
  install(TARGETS peepo_catalogue_live RUNTIME DESTINATION libexec/peepo)
  # Needs an isolated engine and a controlled JACK test tone; not a CTest.
  add_executable(peepo_tuner_live tests/tuner_live.cpp)
  target_link_libraries(peepo_tuner_live PRIVATE peepo_core)
  if(UNIX AND NOT APPLE)
    # Operates on the real Pi patch store only when explicitly invoked.
    add_executable(peepo_patch_live tests/pi_patch_live.cpp)
    target_link_libraries(peepo_patch_live PRIVATE peepo_core)
    install(TARGETS peepo_patch_live RUNTIME DESTINATION libexec/peepo)
    # A conventional effect on the Pi: gain test graph and the combined Clouds patch.
    add_executable(peepo_effect_live tests/pi_effect_live.cpp)
